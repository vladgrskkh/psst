# psst Core Implementation Plan (Spec 1, Part 1 of 2)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the psst core: domain model, bbolt storage, macOS Reminders bridge (Swift + Go client), two-way sync engine, natural date parser, config/theme, and the `psst new` quick-capture CLI — leaving the TUI for plan 2.

**Architecture:** Clean layering `pkg/option + domain ← service ← cli`; `internal/app` composes service ports with infra adapters, while `cmd/psst` stays a thin process entry point. Cobra commands live in `internal/cli`. Sync is a deterministic per-field merge via `RemoteSnapshot`/`LocalSnapshot`; the Reminders bridge is a separate Swift binary speaking line-delimited JSON over stdin/stdout.

**Tech Stack:** Go 1.25.4, bbolt (storage), BurntSushi/toml (config), Cobra (CLI), Swift/EventKit (bridge binary), stdlib `testing` only (no test frameworks).

## Global Constraints

- Module path: `github.com/vladgrskkh/psst`. Go version: `1.25.4`.
- Dependencies: add `github.com/BurntSushi/toml` and `github.com/spf13/cobra`; no test frameworks (stdlib only).

`pkg/option/option.go`:

```go
package option

import (
	"bytes"
	"encoding/json"
)

// Option is a generic wrapper for values that may be absent.
// None serializes to JSON null; Some(v) serializes as v.
type Option[T any] struct {
	value   T
	present bool
}

func Some[T any](v T) Option[T] {
	return Option[T]{value: v, present: true}
}

func None[T any]() Option[T] {
	return Option[T]{}
}

func (o Option[T]) Get() (T, bool) {
	return o.value, o.present
}

func (o Option[T]) IsPresent() bool {
	return o.present
}

func (o Option[T]) ValueOr(fallback T) T {
	if o.present {
		return o.value
	}
	return fallback
}

func (o Option[T]) MarshalJSON() ([]byte, error) {
	if !o.present {
		return []byte("null"), nil
	}
	return json.Marshal(o.value)
}

func (o *Option[T]) UnmarshalJSON(data []byte) error {
	if bytes.Equal(data, []byte("null")) {
		o.present = false
		var zero T
		o.value = zero
		return nil
	}
	if err := json.Unmarshal(data, &o.value); err != nil {
		return err
	}
	o.present = true
	return nil
}
```

package config

import (
	"errors"
	"fmt"
	"os"
	"path/filepath"

	"github.com/BurntSushi/toml"
)

type GeneralConfig struct {
	DefaultList            string `toml:"default_list"`
	SyncIntervalSeconds    int    `toml:"sync_interval_seconds"`
	PushDebounceMs         int    `toml:"push_debounce_ms"`
	TombstoneRetentionDays int    `toml:"tombstone_retention_days"`
	DefaultTime            string `toml:"default_time"`
	DBPath                 string `toml:"db_path"`
	LogLevel               string `toml:"log_level"`
}

type ThemeConfig struct {
	Bg       string `toml:"bg"`
	Surface  string `toml:"surface"`
	Border   string `toml:"border"`
	Text     string `toml:"text"`
	Muted    string `toml:"muted"`
	Accent   string `toml:"accent"`
	Success  string `toml:"success"`
	Warning  string `toml:"warning"`
	Danger   string `toml:"danger"`
	Overdue  string `toml:"overdue"`
	Today    string `toml:"today"`
	Tomorrow string `toml:"tomorrow"`
	Later    string `toml:"later"`
	Cursor   string `toml:"cursor"`
	Progress string `toml:"progress"`
}

type KeysConfig struct {
	Up         []string `toml:"up"`
	Down       []string `toml:"down"`
	ToggleDone []string `toml:"toggle_done"`
	New        []string `toml:"new"`
	Edit       []string `toml:"edit"`
	Delete     []string `toml:"delete"`
	Confirm    []string `toml:"confirm"`
	Cancel     []string `toml:"cancel"`
	ToggleView []string `toml:"toggle_view"`
	Filter     []string `toml:"filter"`
	SyncNow    []string `toml:"sync_now"`
	Quit       []string `toml:"quit"`
}

type Config struct {
	General  GeneralConfig `toml:"general"`
	Theme    ThemeConfig   `toml:"theme"`
	Keys     KeysConfig    `toml:"keys"`
	Warnings []string      `toml:"-"`
}

// Default returns the config with all built-in defaults.
func Default() *Config {
	return &Config{
		General: GeneralConfig{
			DefaultList:            "psst",
			SyncIntervalSeconds:    30,
			PushDebounceMs:         500,
			TombstoneRetentionDays: 30,
			DefaultTime:            "09:00",
			DBPath:                 "",
			LogLevel:               "info",
		},
		Theme: ThemeConfig{
			Bg: "#1e1e2e", Surface: "#313244", Border: "#45475a", Text: "#cdd6f4",
			Muted: "#6c7086", Accent: "#89b4fa", Success: "#a6e3a1", Warning: "#f9e2af",
			Danger: "#f38ba8", Overdue: "#f38ba8", Today: "#fab387", Tomorrow: "#89b4fa",
			Later: "#6c7086", Cursor: "#cba6f7", Progress: "#cba6f7",
		},
		Keys: KeysConfig{
			Up: []string{"k", "up"}, Down: []string{"j", "down"},
			ToggleDone: []string{"space", "x"}, New: []string{"n"}, Edit: []string{"e"},
			Delete: []string{"d"}, Confirm: []string{"y"}, Cancel: []string{"esc"},
			ToggleView: []string{"tab"}, Filter: []string{"/"}, SyncNow: []string{"s"},
			Quit: []string{"q", "ctrl+c"},
		},
	}
}

// Load reads the TOML file at path on top of Default. A missing file returns
// the defaults. Unknown keys are collected in Warnings.
func Load(path string) (*Config, error) {
	cfg := Default()
	if _, err := os.Stat(path); errors.Is(err, os.ErrNotExist) {
		return cfg, nil
	}
	md, err := toml.DecodeFile(path, cfg)
	if err != nil {
		return nil, fmt.Errorf("parse config %s: %w", path, err)
	}
	for _, k := range md.Undecoded() {
		cfg.Warnings = append(cfg.Warnings, fmt.Sprintf("unknown config key: %s", k.String()))
	}
	return cfg, nil
}

// StateDir returns ~/.local/state/psst, creating it if needed.
func StateDir() (string, error) {
	home, err := os.UserHomeDir()
	if err != nil {
		return "", fmt.Errorf("resolve home dir: %w", err)
	}
	dir := filepath.Join(home, ".local", "state", "psst")
	if err := os.MkdirAll(dir, 0o755); err != nil {
		return "", fmt.Errorf("create state dir: %w", err)
	}
	return dir, nil
}

func DefaultDBPath() (string, error) {
	dir, err := StateDir()
	if err != nil {
		return "", err
	}
	return filepath.Join(dir, "psst.db"), nil
}

func DefaultLogPath() (string, error) {
	dir, err := StateDir()
	if err != nil {
		return "", err
	}
	return filepath.Join(dir, "psst.log"), nil
}

### Task 4: Natural date parser

**Files:**
- Create: `internal/dateparser/parser.go` (replace stub), `internal/dateparser/parser_test.go`

**Interfaces:**
- Consumes: nothing.
- Produces:
  - `dateparser.Parse(s string, now time.Time, defaultTime string) (time.Time, error)`
  - `dateparser.ErrInvalidDate` — wrapped with the offending hint: `fmt.Errorf("%w: %q", ErrInvalidDate, s)`.

**Grammar (deterministic, exact):**
1. Trim; empty → ErrInvalidDate.
2. ISO absolute: `2006-01-02` or `2006-01-02 15:04` (time.Parse with layout `"2006-01-02"` then optional `"2006-01-02 15:04"`).
3. Day words: `today`, `tomorrow`; weekdays `mon`/`tue`/`wed`/`thu`/`fri`/`sat`/`sun` (and full names) → the next date with that weekday, where `today`'s weekday counts as "next" only if a time part is also given and that time is later than now; otherwise strictly the next occurrence.
4. Relative: `in 2h` / `in 30m` / `in 2d` / `in 1w` (regex `^in\s+(\d+)\s*([hmdw])$`).
5. Time part (optional): `15:04`, `3:04`, `9am`, `9 am`, `9pm`, `9 pm`, `21:00`. Optional word `at` before it (`tomorrow at 9:00`). Applied to the day part; date-only → time from `defaultTime` (`"09:00"`).
6. Time-only input (no day word): today at that time; if the result is not after `now` → tomorrow.
7. Past results from steps 3-4 (explicit day/relative) are accepted as-is (user asked for them) — no auto-fixing.
8. Anything else → ErrInvalidDate.

// Package dateparser turns natural-language date hints into time.Time values.
package dateparser

import (
	"errors"
	"fmt"
	"regexp"
	"strconv"
	"strings"
	"time"
)

var ErrInvalidDate = errors.New("invalid date hint")

var (
	reISO      = regexp.MustCompile(`^\d{4}-\d{2}-\d{2}$`)
	reISOTime  = regexp.MustCompile(`^\d{4}-\d{2}-\d{2} \d{1,2}:\d{2}$`)
	reRelative = regexp.MustCompile(`^in\s+(\d+)\s*([hmdw])$`)
	reTime     = regexp.MustCompile(`^(\d{1,2}):(\d{2})$`)
	reTimeAMPM = regexp.MustCompile(`^(\d{1,2})\s*(am|pm)$`)
	weekdays   = map[string]time.Weekday{
		"mon": time.Monday, "monday": time.Monday,
		"tue": time.Tuesday, "tuesday": time.Tuesday,
		"wed": time.Wednesday, "wednesday": time.Wednesday,
		"thu": time.Thursday, "thursday": time.Thursday,
		"fri": time.Friday, "friday": time.Friday,
		"sat": time.Saturday, "saturday": time.Saturday,
		"sun": time.Sunday, "sunday": time.Sunday,
	}
)

// Parse converts a natural-language hint into a concrete time.
// now and defaultTime (HH:MM) are injected for determinism.
func Parse(s string, now time.Time, defaultTime string) (time.Time, error) {
	s = strings.TrimSpace(s)
	if s == "" {
		return time.Time{}, invalid(s)
	}
	if reISO.MatchString(s) {
		return withDefaultTime(s, "2006-01-02", defaultTime, now)
	}
	if reISOTime.MatchString(s) {
		t, err := time.ParseInLocation("2006-01-02 15:04", s, time.Local)
		if err != nil {
			return time.Time{}, invalid(s)
		}
		return t, nil
	}

	if m := reRelative.FindStringSubmatch(s); m != nil {
		n, err := strconv.Atoi(m[1])
		if err != nil {
			return time.Time{}, invalid(s)
		}
		switch m[2] {
		case "h":
			return now.Add(time.Duration(n) * time.Hour), nil
		case "m":
			return now.Add(time.Duration(n) * time.Minute), nil
		case "d":
			return now.AddDate(0, 0, n), nil
		case "w":
			return now.AddDate(0, 0, 7*n), nil
		}
	}

	fields := strings.Fields(s)
	head := strings.ToLower(fields[0])

	// Bare time: today at that time, rolled to tomorrow if already past.
	if len(fields) == 1 {
		if _, err := parseHM(head); err == nil {
			return timeOnly(head, now)
		}
	}

	var day time.Time
	haveDay := false
	switch head {
	case "today":
		day = midnight(now)
		haveDay = true
	case "tomorrow":
		day = midnight(now).AddDate(0, 0, 1)
		haveDay = true
	default:
		if wd, ok := weekdays[head]; ok {
			delta := (int(wd) - int(now.Weekday()) + 7) % 7
			day = midnight(now).AddDate(0, 0, delta)
			haveDay = true
		}
	}
	if !haveDay {
		return time.Time{}, invalid(s)
	}

	rest := fields[1:]
	if len(rest) > 0 && strings.EqualFold(rest[0], "at") {
		rest = rest[1:]
	}
	timeStr := defaultTime
	if len(rest) > 0 {
		timeStr = strings.Join(rest, "")
	}
	hm, err := parseHM(timeStr)
	if err != nil {
		return time.Time{}, invalid(s)
	}
	t := time.Date(day.Year(), day.Month(), day.Day(), hm[0], hm[1], 0, 0, now.Location())

	// "today" (or same-weekday) with a time that's already past → next week.
	if day.Equal(midnight(now)) && !t.After(now) && len(rest) > 0 {
		t = t.AddDate(0, 0, 7)
	}
	return t, nil
}

func midnight(now time.Time) time.Time {
	return time.Date(now.Year(), now.Month(), now.Day(), 0, 0, 0, 0, now.Location())
}

func timeOnly(s string, now time.Time) (time.Time, error) {
	hm, err := parseHM(s)
	if err != nil {
		return time.Time{}, invalid(s)
	}
	t := time.Date(now.Year(), now.Month(), now.Day(), hm[0], hm[1], 0, 0, now.Location())
	if !t.After(now) {
		t = t.AddDate(0, 0, 1)
	}
	return t, nil
}

func parseHM(s string) ([2]int, error) {
	if m := reTime.FindStringSubmatch(s); m != nil {
		h, _ := strconv.Atoi(m[1])
		min, _ := strconv.Atoi(m[2])
		if h > 23 || min > 59 {
			return [2]int{}, fmt.Errorf("bad time")
		}
		return [2]int{h, min}, nil
	}
	if m := reTimeAMPM.FindStringSubmatch(s); m != nil {
		h, _ := strconv.Atoi(m[1])
		if h == 0 || h > 12 {
			return [2]int{}, fmt.Errorf("bad time")
		}
		if m[2] == "pm" && h != 12 {
			h += 12
		}
		if m[2] == "am" && h == 12 {
			h = 0
		}
		return [2]int{h, 0}, nil
	}
	return [2]int{}, fmt.Errorf("bad time")
}

func withDefaultTime(s, layout, defaultTime string, now time.Time) (time.Time, error) {
	t, err := time.ParseInLocation(layout, s, time.Local)
	if err != nil {
		return time.Time{}, invalid(s)
	}
	hm, err := parseHM(defaultTime)
	if err != nil {
		return time.Time{}, invalid(s)
	}
	return time.Date(t.Year(), t.Month(), t.Day(), hm[0], hm[1], 0, 0, now.Location()), nil
}

func invalid(s string) error {
	return fmt.Errorf("%w: %q", ErrInvalidDate, s)
}

// Package store persists todos in a local bbolt database.
package store

import (
	"encoding/json"
	"errors"
	"fmt"
	"sort"
	"time"

	"github.com/vladgrskkh/psst/internal/domain"
	"github.com/vladgrskkh/psst/internal/service"
	bolt "go.etcd.io/bbolt"
)

var (
	ErrNotFound      = errors.New("todo not found")
	bucketTodos      = []byte("todos")
	bucketMeta       = []byte("meta")
	keySyncState     = []byte("sync_state")
)

func Open(path string) (*bolt.DB, error) {
	db, err := bolt.Open(path, 0o600, nil)
	if err != nil {
		return nil, fmt.Errorf("open bbolt db: %w", err)
	}
	err = db.Update(func(tx *bolt.Tx) error {
		for _, name := range [][]byte{bucketTodos, bucketMeta} {
			if _, err := tx.CreateBucketIfNotExists(name); err != nil {
				return err
			}
		}
		return nil
	})
	if err != nil {
		db.Close()
		return nil, fmt.Errorf("create buckets: %w", err)
	}
	return db, nil
}

type TodoStore struct {
	db *bolt.DB
}

func NewTodoStore(db *bolt.DB) *TodoStore {
	return &TodoStore{db: db}
}

func (s *TodoStore) Get(id string) (*domain.Todo, error) {
	var todo *domain.Todo
	err := s.db.View(func(tx *bolt.Tx) error {
		b := tx.Bucket(bucketTodos)
		data := b.Get([]byte(id))
		if data == nil {
			return ErrNotFound
		}
		var t domain.Todo
		if err := json.Unmarshal(data, &t); err != nil {
			return fmt.Errorf("decode todo %s: %w", id, err)
		}
		todo = &t
		return nil
	})
	if err != nil {
		return nil, err
	}
	return todo, nil
}

func (s *TodoStore) List() ([]*domain.Todo, error) {
	todos := make([]*domain.Todo, 0)
	err := s.db.View(func(tx *bolt.Tx) error {
		b := tx.Bucket(bucketTodos)
		return b.ForEach(func(_, v []byte) error {
			var t domain.Todo
			if err := json.Unmarshal(v, &t); err != nil {
				return fmt.Errorf("decode todo: %w", err)
			}
			todos = append(todos, &t)
			return nil
		})
	})
	if err != nil {
		return nil, err
	}
	sort.Slice(todos, func(i, j int) bool { return todos[i].CreatedAt.Before(todos[j].CreatedAt) })
	return todos, nil
}

func (s *TodoStore) ListByReminderID(rid string) (*domain.Todo, error) {
	todos, err := s.List()
	if err != nil {
		return nil, err
	}
	for _, t := range todos {
		if t.ReminderID.IsPresent() && t.ReminderID.ValueOr("") == rid {
			return t, nil
		}
	}
	return nil, ErrNotFound
}

func (s *TodoStore) Put(todo *domain.Todo) error {
	data, err := json.Marshal(todo)
	if err != nil {
		return fmt.Errorf("encode todo: %w", err)
	}
	return s.db.Update(func(tx *bolt.Tx) error {
		return tx.Bucket(bucketTodos).Put([]byte(todo.ID), data)
	})
}

func (s *TodoStore) PurgeTombstones(cutoff time.Time, lastPushAt time.Time) error {
	return s.db.Update(func(tx *bolt.Tx) error {
		b := tx.Bucket(bucketTodos)
		var toDelete [][]byte
		err := b.ForEach(func(k, v []byte) error {
			var t domain.Todo
			if err := json.Unmarshal(v, &t); err != nil {
				return err
			}
			if t.Deleted && t.DeletedAt.IsPresent() &&
				t.DeletedAt.ValueOr(time.Time{}).Before(cutoff) &&
				!t.UpdatedAt.After(lastPushAt) {
				toDelete = append(toDelete, k)
			}
			return nil
		})
		if err != nil {
			return err
		}
		for _, k := range toDelete {
			if err := b.Delete(k); err != nil {
				return err
			}
		}
		return nil
	})
}

func (s *TodoStore) GetSyncState() (domain.SyncState, error) {
	var state domain.SyncState
	err := s.db.View(func(tx *bolt.Tx) error {
		data := tx.Bucket(bucketMeta).Get(keySyncState)
		if data == nil {
			return nil
		}
		return json.Unmarshal(data, &state)
	})
	return state, err
}

func (s *TodoStore) SaveSyncState(state domain.SyncState) error {
	data, err := json.Marshal(state)
	if err != nil {
		return err
	}
	return s.db.Update(func(tx *bolt.Tx) error {
		return tx.Bucket(bucketMeta).Put(keySyncState, data)
	})
}

package service

import (
	"errors"
	"strings"
	"time"

	"github.com/vladgrskkh/psst/internal/domain"
)

var (
	ErrNotFound       = errors.New("todo not found")
	ErrInvalidTitle   = errors.New("title must not be empty")
	ErrDeadlineInPast = errors.New("deadline must be in the future")
	ErrInvalidList    = errors.New("list name must not be empty")
)

type TodoService struct {
	store TodoStore
	clock Clock
}

func NewTodoService(store TodoStore, clock Clock) *TodoService {
	return &TodoService{store: store, clock: clock}
}

type CreateTodoParams struct {
	Title       string
	Description string
	Deadline    option.Option[time.Time]
	ListName    string
}

func (s *TodoService) Create(p CreateTodoParams) (*domain.Todo, error) {
	title := strings.TrimSpace(p.Title)
	list := strings.TrimSpace(p.ListName)
	if err := validate(title, list, p.Deadline, s.clock.Now()); err != nil {
		return nil, err
	}
	now := s.clock.Now()
	todo := domain.NewTodo(title)
	todo.Description = strings.TrimSpace(p.Description)
	todo.Deadline = p.Deadline
	todo.ListName = list
	todo.CreatedAt = now
	todo.UpdatedAt = now
	if err := s.store.Put(todo); err != nil {
		return nil, err
	}
	return todo, nil
}

type UpdateTodoParams struct {
	Title       option.Option[string]
	Description option.Option[string]
	Deadline    option.Option[option.Option[time.Time]]
	ListName    option.Option[string]
}

func (s *TodoService) Update(id string, p UpdateTodoParams) (*domain.Todo, error) {
	todo, err := s.store.Get(id)
	if err != nil {
		return nil, err
	}
	if p.Title.IsPresent() {
		todo.Title = strings.TrimSpace(p.Title.ValueOr(""))
	}
	if p.Description.IsPresent() {
		todo.Description = strings.TrimSpace(p.Description.ValueOr(""))
	}
	if p.Deadline.IsPresent() {
		todo.Deadline = p.Deadline.ValueOr(option.None[time.Time]())
	}
	if p.ListName.IsPresent() {
		todo.ListName = strings.TrimSpace(p.ListName.ValueOr(""))
	}
	if err := validate(todo.Title, todo.ListName, todo.Deadline, s.clock.Now()); err != nil {
		return nil, err
	}
	todo.UpdatedAt = s.clock.Now()
	if err := s.store.Put(todo); err != nil {
		return nil, err
	}
	return todo, nil
}

func (s *TodoService) ToggleDone(id string) (*domain.Todo, error) {
	todo, err := s.store.Get(id)
	if err != nil {
		return nil, err
	}
	now := s.clock.Now()
	todo.Done = !todo.Done
	if todo.Done {
		todo.CompletedAt = option.Some(now)
	} else {
		todo.CompletedAt = option.None[time.Time]()
	}
	todo.UpdatedAt = now
	if err := s.store.Put(todo); err != nil {
		return nil, err
	}
	return todo, nil
}

func (s *TodoService) Delete(id string) error {
	todo, err := s.store.Get(id)
	if err != nil {
		return err
	}
	now := s.clock.Now()
	todo.Deleted = true
	todo.DeletedAt = option.Some(now)
	todo.UpdatedAt = now
	return s.store.Put(todo)
}

func (s *TodoService) List(filter string) ([]*domain.Todo, error) {
	all, err := s.store.List()
	if err != nil {
		return nil, err
	}
	filter = strings.ToLower(strings.TrimSpace(filter))
	out := make([]*domain.Todo, 0, len(all))
	for _, t := range all {
		if t.Deleted {
			continue
		}
		if filter != "" && !strings.Contains(strings.ToLower(t.Title), filter) &&
			!strings.Contains(strings.ToLower(t.Description), filter) {
			continue
		}
		out = append(out, t)
	}
	return out, nil
}

func validate(title, list string, deadline option.Option[time.Time], now time.Time) error {
	if title == "" {
		return ErrInvalidTitle
	}
	if list == "" {
		return ErrInvalidList
	}
	if deadline.IsPresent() && !deadline.ValueOr(time.Time{}).After(now) {
		return ErrDeadlineInPast
	}
	return nil
}

import Foundation
import EventKit

let eventStore = EKEventStore()
let dateFormatter = DateFormatter()
dateFormatter.locale = Locale(identifier: "en_US_POSIX")
dateFormatter.dateFormat = "yyyy-MM-dd'T'HH:mm:ssZ"

func send(_ id: Int, result: [String: Any]) {
    let out: [String: Any] = ["id": id, "result": result]
    if let data = try? JSONSerialization.data(withJSONObject: out),
       let str = String(data: data, encoding: .utf8) {
        print(str)
        fflush(stdout)
    }
}

func sendError(_ id: Int, code: String, message: String) {
    let out: [String: Any] = ["id": id, "error": ["code": code, "message": message]]
    if let data = try? JSONSerialization.data(withJSONObject: out),
       let str = String(data: data, encoding: .utf8) {
        print(str)
        fflush(stdout)
    }
}

func dateToString(_ date: Date?) -> Any {
    guard let date = date else { return NSNull() }
    return dateFormatter.string(from: date)
}

func stringToDate(_ s: String) -> Date? {
    return dateFormatter.date(from: s)
}

func requestAccessIfNeeded() {
    if EKEventStore.authorizationStatus(for: .reminder) == .notDetermined {
        let sem = DispatchSemaphore(value: 0)
        eventStore.requestAccess(to: .reminder) { _, _ in sem.signal() }
        sem.wait()
    }
}

func permissionStatus() -> String {
    switch EKEventStore.authorizationStatus(for: .reminder) {
    case .authorized: return "authorized"
    case .notDetermined: return "not-determined"
    default: return "denied"
    }
}

func findReminder(_ id: String) -> EKReminder? {
    for cal in eventStore.calendars(for: .reminder) {
        let predicate = eventStore.predicateForReminders(in: [cal])
        let sem = DispatchSemaphore(value: 0)
        var found: EKReminder?
        eventStore.fetchReminders(matching: predicate) { reminders in
            found = reminders?.first(where: { $0.calendarItemIdentifier == id })
            sem.signal()
        }
        sem.wait()
        if found != nil { return found }
    }
    return nil
}

func handle(_ id: Int, _ method: String, _ params: [String: Any]) {
    switch method {
    case "permissions":
        requestAccessIfNeeded()
        send(id, result: ["status": permissionStatus()])

    case "list-lists":
        var lists: [[String: Any]] = []
        for cal in eventStore.calendars(for: .reminder) {
            lists.append(["id": cal.calendarIdentifier, "name": cal.title])
        }
        send(id, result: ["lists": lists])

    case "ensure-list":
        let name = params["name"] as? String ?? ""
        if name.isEmpty {
            sendError(id, code: "bad-params", message: "name required")
            return
        }
        if let existing = eventStore.calendars(for: .reminder).first(where: { $0.title == name }) {
            send(id, result: ["id": existing.calendarIdentifier])
            return
        }
        guard let cal = EKCalendar(for: .reminder, eventStore: eventStore) else {
            sendError(id, code: "create-failed", message: "cannot create calendar")
            return
        }
        cal.title = name
        do {
            try eventStore.saveCalendar(cal, commit: true)
            send(id, result: ["id": cal.calendarIdentifier])
        } catch {
            sendError(id, code: "create-failed", message: error.localizedDescription)
        }

    case "pull":
        let listIDs = params["listIDs"] as? [String] ?? []
        var reminders: [[String: Any]] = []
        var seen = Set<String>()
        let cals = eventStore.calendars(for: .reminder).filter { listIDs.contains($0.calendarIdentifier) }
        for cal in cals {
            let predicate = eventStore.predicateForReminders(in: [cal])
            let sem = DispatchSemaphore(value: 0)
            eventStore.fetchReminders(matching: predicate) { fetched in
                for r in fetched ?? [] where !seen.contains(r.calendarItemIdentifier) {
                    seen.insert(r.calendarItemIdentifier)
                    reminders.append([
                        "id": r.calendarItemIdentifier,
                        "listID": cal.calendarIdentifier,
                        "title": r.title ?? "",
                        "notes": r.notes ?? "",
                        "dueDate": dateToString(r.dueDateComponents.map {
                            Calendar.current.date(from: $0) ?? Date()
                        } ?? nil),
                        "completed": r.isCompleted,
                        "completionDate": dateToString(r.completionDate),
                    ])
                }
                sem.signal()
            }
            sem.wait()
        }
        send(id, result: ["reminders": reminders])

    case "create":
        guard let listID = params["listID"] as? String,
              let cal = eventStore.calendars(for: .reminder).first(where: { $0.calendarIdentifier == listID }) else {
            sendError(id, code: "bad-params", message: "listID required")
            return
        }
        let reminder = EKReminder(eventStore: eventStore)
        reminder.title = params["title"] as? String ?? ""
        reminder.notes = params["notes"] as? String ?? ""
        reminder.calendar = cal
        if let due = params["dueDate"] as? String, let d = stringToDate(due) {
            reminder.dueDateComponents = Calendar.current.dateComponents(
                [.year, .month, .day, .hour, .minute], from: d)
        }
        do {
            try eventStore.save(reminder, commit: true)
            send(id, result: ["id": reminder.calendarItemIdentifier])
        } catch {
            sendError(id, code: "save-failed", message: error.localizedDescription)
        }

    case "update":
        guard let rid = params["id"] as? String,
              let reminder = findReminder(rid) else {
            sendError(id, code: "not-found", message: "reminder not found")
            return
        }
        let fields = params["fields"] as? [String: Any] ?? [:]
        if let title = fields["title"] as? String { reminder.title = title }
        if let notes = fields["notes"] as? String { reminder.notes = notes }
        if fields["dueDate"] is NSNull || fields["dueDate"] == nil {
            // no change when key absent; handled below
        }
        if let due = fields["dueDate"] as? String, let d = stringToDate(due) {
            reminder.dueDateComponents = Calendar.current.dateComponents(
                [.year, .month, .day, .hour, .minute], from: d)
        }
        if let completed = fields["completed"] as? Bool {
            reminder.isCompleted = completed
            if completed {
                reminder.completionDate = (fields["completionDate"] as? String).flatMap(stringToDate) ?? Date()
            } else {
                reminder.completionDate = nil
            }
        }
        do {
            try eventStore.save(reminder, commit: true)
            send(id, result: [:])
        } catch {
            sendError(id, code: "save-failed", message: error.localizedDescription)
        }

    case "delete":
        guard let rid = params["id"] as? String,
              let reminder = findReminder(rid) else {
            sendError(id, code: "not-found", message: "reminder not found")
            return
        }
        do {
            try eventStore.remove(reminder, commit: true)
            send(id, result: [:])
        } catch {
            sendError(id, code: "delete-failed", message: error.localizedDescription)
        }

    default:
        sendError(id, code: "unknown-method", message: method)
    }
}

while let line = readLine() {
    guard let data = line.data(using: .utf8),
          let obj = try? JSONSerialization.jsonObject(with: data) as? [String: Any],
          let id = obj["id"] as? Int,
          let method = obj["method"] as? String else {
        continue
    }
    handle(id, method, obj["params"] as? [String: Any] ?? [:])

// line-delimited JSON on stdin/stdout.
package reminders

import (
	"bufio"
	"bytes"
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"log/slog"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
	"time"

	"github.com/vladgrskkh/psst/internal/domain"
	"github.com/vladgrskkh/psst/internal/service"
)

var (
	ErrBridgeNotFound   = errors.New("psst-bridge binary not found — run `make build/bridge`")
	ErrAccessDenied     = errors.New("reminders access denied — enable in System Settings → Privacy & Security → Reminders")
	ErrReminderNotFound = errors.New("reminder not found")
	ErrBridgeProtocol   = errors.New("unexpected bridge response")
)

type Client struct {
	binPath string
	timeout time.Duration
	logger  *slog.Logger
}

// NewClient resolves the bridge binary next to the psst executable.
func NewClient(logger *slog.Logger) *Client {
	exe, err := os.Executable()
	if err != nil {
		exe = "."
	}
	return newClientAt(filepath.Join(filepath.Dir(exe), "psst-bridge"), 15*time.Second, logger)
}

func newClientAt(binPath string, timeout time.Duration, logger *slog.Logger) *Client {
	return &Client{binPath: binPath, timeout: timeout, logger: logger}
}

type bridgeResponse struct {
	ID     int             `json:"id"`
	Result json.RawMessage `json:"result"`
	Error  *bridgeError    `json:"error"`
}

type bridgeError struct {
	Code    string `json:"code"`
	Message string `json:"message"`
}

// call runs one request against a fresh bridge process.
func (c *Client) call(method string, params map[string]any) (json.RawMessage, error) {
	ctx, cancel := context.WithTimeout(context.Background(), c.timeout)
	defer cancel()

	cmd := exec.CommandContext(ctx, c.binPath)
	stdin, err := cmd.StdinPipe()
	if err != nil {
		return nil, fmt.Errorf("bridge stdin: %w", err)
	}
	stdout, err := cmd.StdoutPipe()
	if err != nil {
		return nil, fmt.Errorf("bridge stdout: %w", err)
	}
	if err := cmd.Start(); err != nil {
		if ctx.Err() != nil {
			return nil, fmt.Errorf("%w (timed out)", ctx.Err())
		}
		return nil, fmt.Errorf("%w: %v", ErrBridgeNotFound, err)
	}
	defer cmd.Wait() //nolint:errcheck

	req := map[string]any{"id": 1, "method": method}
	if len(params) > 0 {
		req["params"] = params
	}
	var buf bytes.Buffer
	if err := json.NewEncoder(&buf).Encode(req); err != nil {
		return nil, err
	}
	if _, err := stdin.Write(buf.Bytes()); err != nil {
		return nil, fmt.Errorf("write request: %w", err)
	}
	stdin.Close()

	var resp bridgeResponse
	if err := json.NewDecoder(bufio.NewReader(stdout)).Decode(&resp); err != nil {
		return nil, fmt.Errorf("%w: decode response: %v", ErrBridgeProtocol, err)
	}
	if resp.ID != 1 {
		return nil, fmt.Errorf("%w: id mismatch", ErrBridgeProtocol)
	}
	if resp.Error != nil {
		return nil, mapBridgeError(resp.Error)
	}
	return resp.Result, nil
}

func mapBridgeError(e *bridgeError) error {
	switch e.Code {
	case "access-denied":
		return ErrAccessDenied
	case "not-found":
		return ErrReminderNotFound
	default:
		return fmt.Errorf("bridge error (%s): %s", e.Code, e.Message)
	}
}

func (c *Client) Permissions() (string, error) {
	raw, err := c.call("permissions", nil)
	if err != nil {
		return "", err
	}
	var out struct {
		Status string `json:"status"`
	}
	if err := json.Unmarshal(raw, &out); err != nil {
		return "", fmt.Errorf("%w: %v", ErrBridgeProtocol, err)
	}
	switch out.Status {
	case "authorized", "denied", "not-determined":
		return out.Status, nil
	}
	return "", fmt.Errorf("%w: unknown status %q", ErrBridgeProtocol, out.Status)
}

func (c *Client) ListLists() ([]service.ReminderList, error) {
	raw, err := c.call("list-lists", nil)
	if err != nil {
		return nil, err
	}
	var out struct {
		Lists []struct {
			ID   string `json:"id"`
			Name string `json:"name"`
		} `json:"lists"`
	}
	if err := json.Unmarshal(raw, &out); err != nil {
		return nil, fmt.Errorf("%w: %v", ErrBridgeProtocol, err)
	}
	lists := make([]service.ReminderList, 0, len(out.Lists))
	for _, l := range out.Lists {
		lists = append(lists, service.ReminderList{ID: l.ID, Name: l.Name})
	}
	return lists, nil
}

func (c *Client) EnsureList(name string) (service.ReminderList, error) {
	raw, err := c.call("ensure-list", map[string]any{"name": name})
	if err != nil {
		return service.ReminderList{}, err
	}
	var out struct {
		ID string `json:"id"`
	}
	if err := json.Unmarshal(raw, &out); err != nil {
		return service.ReminderList{}, fmt.Errorf("%w: %v", ErrBridgeProtocol, err)
	}
	return service.ReminderList{ID: out.ID, Name: name}, nil
}

func (c *Client) Pull(listIDs []string) ([]service.Reminder, error) {
	raw, err := c.call("pull", map[string]any{"listIDs": listIDs})
	if err != nil {
		return nil, err
	}
	var out struct {
		Reminders []struct {
			ID             string  `json:"id"`
			ListID         string  `json:"listID"`
			Title          string  `json:"title"`
			Notes          string  `json:"notes"`
			DueDate        *string `json:"dueDate"`
			Completed      bool    `json:"completed"`
			CompletionDate *string `json:"completionDate"`
		} `json:"reminders"`
	}
	if err := json.Unmarshal(raw, &out); err != nil {
		return nil, fmt.Errorf("%w: %v", ErrBridgeProtocol, err)
	}
	reminders := make([]service.Reminder, 0, len(out.Reminders))
	for _, r := range out.Reminders {
		reminders = append(reminders, service.Reminder{
			ID:             r.ID,
			ListID:         r.ListID,
			Title:          r.Title,
			Notes:          r.Notes,
			DueDate:        parseOptTime(r.DueDate),
			Completed:      r.Completed,
			CompletionDate: parseOptTime(r.CompletionDate),
		})
	}
	return reminders, nil
}

func (c *Client) Create(p service.CreateReminderParams) (string, error) {
	params := map[string]any{
		"listID": p.ListID,
		"title":  p.Title,
		"notes":  p.Notes,
	}
	if p.DueDate.IsPresent() {
		params["dueDate"] = p.DueDate.ValueOr(time.Time{}).Format(time.RFC3339)
	} else {
		params["dueDate"] = nil
	}
	raw, err := c.call("create", params)
	if err != nil {
		return "", err
	}
	var out struct {
		ID string `json:"id"`
	}
	if err := json.Unmarshal(raw, &out); err != nil {
		return "", fmt.Errorf("%w: %v", ErrBridgeProtocol, err)
	}
	return out.ID, nil
}

func (c *Client) Update(p service.UpdateReminderParams) error {
	fields := map[string]any{}
	if p.Title.IsPresent() {
		fields["title"] = p.Title.ValueOr("")
	}
	if p.Notes.IsPresent() {
		fields["notes"] = p.Notes.ValueOr("")
	}
	if p.ListID.IsPresent() {
		fields["listID"] = p.ListID.ValueOr("")
	}
	if p.DueDate.IsPresent() {
		d := p.DueDate.ValueOr(option.None[time.Time]())
		if d.IsPresent() {
			fields["dueDate"] = d.ValueOr(time.Time{}).Format(time.RFC3339)
		} else {
			fields["dueDate"] = nil
		}
	}
	if p.Completed.IsPresent() {
		fields["completed"] = p.Completed.ValueOr(false)
	}
	if p.CompletionDate.IsPresent() {
		d := p.CompletionDate.ValueOr(option.None[time.Time]())
		if d.IsPresent() {
			fields["completionDate"] = d.ValueOr(time.Time{}).Format(time.RFC3339)
		} else {
			fields["completionDate"] = nil
		}
	}
	_, err := c.call("update", map[string]any{"id": p.ID, "fields": fields})
	return err
}

func (c *Client) Delete(id string) error {
	_, err := c.call("delete", map[string]any{"id": id})
	return err
}

func parseOptTime(s *string) option.Option[time.Time] {
	if s == nil {
		return option.None[time.Time]()
	}
	t, err := time.Parse(time.RFC3339, strings.TrimSpace(*s))
	if err != nil {
		return option.None[time.Time]()
	}
	return option.Some(t)
}

**Files:**
- Create: `internal/service/sync.go` (replace stub), `internal/service/sync_push_test.go`

**Interfaces:**
- Consumes: ports from Task 6, `domain` types.
- Produces (shared with Task 9/10):
  - `service.NewSyncEngine(store TodoStore, meta SyncMeta, gateway RemindersGateway, clock Clock, defaultListName string, tombstoneRetention time.Duration) *SyncEngine`
  - `(*SyncEngine) FullSync() error`, `(*SyncEngine) PushChanges() error`, `(*SyncEngine) PullChanges() error`
  - `service.Syncer` interface: `FullSync() error; PushChanges() error; PullChanges() error`
  - Snapshot model (used by store as opaque strings): `ReminderState{ Title, Notes string; DueDate option.Option[time.Time]; ListID string; Completed bool; CompletionDate option.Option[time.Time]; Deleted bool }` with helpers `reminderStateFromTodo(*domain.Todo) ReminderState`, `snapshotJSON(ReminderState) string`, `parseSnapshot(string) (ReminderState, error)` (empty string → zero state, no error), `diffStates(a, b ReminderState) map[string]bool`.

**Push behavior (exact):**
1. Read `SyncState`; `lastPushAt = state.LastPushAt`.
2. `needsPush(t)` = `!t.ReminderID.IsPresent() && !t.Deleted` OR `diffStates(parseSnapshot(t.LocalSnapshot), reminderStateFromTodo(t))` non-empty. (Empty/absent LocalSnapshot diffs as changed.)
3. Ensure the default list exists via `gateway.EnsureList(defaultListName)` — cache per call.
4. For each todo needing push (in CreatedAt order):
   - If `t.Deleted` and ReminderID present → `gateway.Delete(rid)`.
   - If ReminderID absent → resolve list: `t.ListID` if present, else `EnsureList(t.ListName)`; `gateway.Create(...)`; set `t.ReminderID`, `t.ListID`.
   - Else → `gateway.Update` with all fields as `Some` values (full-state update).
   - On `ErrReminderNotFound` from Update → fall back to Create (fresh ReminderID).
   - Success → `t.RemoteSnapshot = snapshotJSON(state)`, `t.LocalSnapshot = snapshotJSON(state)`; `store.Put(t)`.
5. Save `SyncState{LastPushAt: clock.Now(), LastPullAt: state.LastPullAt}`.
6. Purge tombstones: `store.PurgeTombstones(clock.Now().Add(-retention), clock.Now())`.

func parseSnapshot(s string) (ReminderState, error) {
	if s == "" {
		return ReminderState{}, nil

	var state ReminderState
	if err := json.Unmarshal([]byte(s), &state); err != nil {
		return ReminderState{}, err

// diffStates returns the names of fields that differ between a and b.
func diffStates(a, b ReminderState) map[string]bool {
	diff := map[string]bool{}
	aJ, _ := json.Marshal(a)
	bJ, _ := json.Marshal(b)
	var aM, bM map[string]json.RawMessage
	json.Unmarshal(aJ, &aM) //nolint:errcheck
	json.Unmarshal(bJ, &bM) //nolint:errcheck
	for _, f := range syncFields {
		if string(aM[f]) != string(bM[f]) {
			diff[f] = true
		}
	}
	return diff

func (e *SyncEngine) needsPush(t *domain.Todo) bool {
	if !t.ReminderID.IsPresent() && !t.Deleted {
		return true
	}
	local, err := parseSnapshot(t.LocalSnapshot)
	if err != nil {
		return true
	}
	return len(diffStates(local, reminderStateFromTodo(t))) > 0
}

// PushChanges pushes locally-changed todos to Reminders.
func (e *SyncEngine) PushChanges() error {
	state, err := e.meta.GetSyncState()
	if err != nil {
		return err
	}
	todos, err := e.store.List()
	if err != nil {
		return err
	}

	for _, t := range todos {
		if !e.needsPush(t) {
			continue
		}
		if t.Deleted && t.ReminderID.IsPresent() {
			if err := e.gateway.Delete(t.ReminderID.ValueOr("")); err != nil {
				return err
			}
			st := reminderStateFromTodo(t)
			t.RemoteSnapshot = snapshotJSON(st)
			t.LocalSnapshot = snapshotJSON(st)
			if err := e.store.Put(t); err != nil {
				return err
			}
			continue
		}

		if !t.ReminderID.IsPresent() {
			listID, err := e.resolveListID(t)
			if err != nil {
				return err
			}
			rid, err := e.gateway.Create(CreateReminderParams{
				ListID:  listID,
				Title:   t.Title,
				Notes:   t.Description,
				DueDate: t.Deadline,
			})
			if err != nil {
				return err
			}
			t.ReminderID = option.Some(rid)
		} else {
			err := e.gateway.Update(e.fullUpdateParams(t))
			if errors.Is(err, ErrReminderNotFound) {
				rid, createErr := e.createReminder(t)
				if createErr != nil {
					return createErr
				}
				t.ReminderID = option.Some(rid)
			} else if err != nil {
				return err
			}
		}

		st := reminderStateFromTodo(t)
		t.RemoteSnapshot = snapshotJSON(st)
		t.LocalSnapshot = snapshotJSON(st)
		if err := e.store.Put(t); err != nil {
			return err
		}
	}

	state.LastPushAt = e.clock.Now()
	if err := e.meta.SaveSyncState(state); err != nil {
		return err
	}
	now := e.clock.Now()
	return e.store.PurgeTombstones(now.Add(-e.tombstoneRetention), now)
}

// PullChanges fetches Reminders state and merges it into the local store.
func (e *SyncEngine) PullChanges() error {
	state, err := e.meta.GetSyncState()
	if err != nil {
		return err
	}
	lists, err := e.gateway.ListLists()
	if err != nil {
		return err
	}
	nameByID := map[string]string{}
	for _, l := range lists {
		nameByID[l.ID] = l.Name
	}
	defaultListID := ""
	for _, l := range lists {
		if l.Name == e.defaultListName {
			defaultListID = l.ID
		}
	}
	if defaultListID == "" {
		list, err := e.gateway.EnsureList(e.defaultListName)
		if err != nil {
			return err
		}
		defaultListID = list.ID
		nameByID[defaultListID] = e.defaultListName
	}

	todos, err := e.store.List()
	if err != nil {
		return err
	}
	tracked := map[string]bool{defaultListID: true}
	for _, t := range todos {
		if t.ListID.IsPresent() {
			tracked[t.ListID.ValueOr("")] = true
		}
	}
	trackedIDs := make([]string, 0, len(tracked))
	for id := range tracked {
		trackedIDs = append(trackedIDs, id)
	}

	reminders, err := e.gateway.Pull(trackedIDs)
	if err != nil {
		return err
	}
	byID := map[string]Reminder{}
	for _, r := range reminders {
		byID[r.ID] = r
	}

	now := e.clock.Now()
	for _, t := range todos {
		if !t.ReminderID.IsPresent() {
			continue
		}
		rid := t.ReminderID.ValueOr("")
		r, found := byID[rid]
		if !found {
			prevLocal, _ := parseSnapshot(t.LocalSnapshot)
			local := reminderStateFromTodo(t)
			if !t.Deleted && len(diffStates(prevLocal, local)) == 0 {
				t.Deleted = true
				t.DeletedAt = option.Some(now)
				t.UpdatedAt = now
				st := reminderStateFromTodo(t)
				t.RemoteSnapshot = snapshotJSON(st)
				t.LocalSnapshot = snapshotJSON(st)
				if err := e.store.Put(t); err != nil {
					return err
				}
			}
			continue
		}

		prevRemote, _ := parseSnapshot(t.RemoteSnapshot)
		prevLocal, _ := parseSnapshot(t.LocalSnapshot)
		remote := reminderStateFromReminder(r)
		local := reminderStateFromTodo(t)
		remoteChanged := diffStates(prevRemote, remote)
		localChanged := diffStates(prevLocal, local)
		for field := range remoteChanged {
			if !localChanged[field] {
				applyField(t, field, remote)
			}
		}
		if name, ok := nameByID[remote.ListID]; ok && t.ListName == "" {
			t.ListName = name
		}
		t.UpdatedAt = now
		t.RemoteSnapshot = snapshotJSON(remote)
		t.LocalSnapshot = snapshotJSON(reminderStateFromTodo(t))
		if err := e.store.Put(t); err != nil {
			return err
		}
		delete(byID, rid)
	}

	for _, r := range byID {
		todo := domain.NewTodo(r.Title)
		todo.ReminderID = option.Some(r.ID)
		todo.ListID = option.Some(r.ListID)
		todo.ListName = nameByID[r.ListID]
		if todo.ListName == "" {
			todo.ListName = "Reminders"
		}
		todo.CreatedAt = now
		todo.UpdatedAt = now
		applyReminderState(todo, reminderStateFromReminder(r))
		todo.RemoteSnapshot = snapshotJSON(reminderStateFromReminder(r))
		todo.LocalSnapshot = snapshotJSON(reminderStateFromTodo(todo))
		if err := e.store.Put(todo); err != nil {
			return err
		}
	}

	state.LastPullAt = now
	return e.meta.SaveSyncState(state)
}

package service

import (
	"sync"
	"time"

	"github.com/vladgrskkh/psst/internal/domain"
)

type SyncStatus struct {
	Syncing      bool
	LastSyncedAt option.Option[time.Time]
	LastError    error
}

type SyncService struct {
	engine   Syncer
	debounce time.Duration

	mu     sync.Mutex
	status SyncStatus
	timer  *time.Timer
}

func NewSyncService(engine Syncer, debounce time.Duration) *SyncService {
	return &SyncService{engine: engine, debounce: debounce}
}

// SyncNow runs a full sync synchronously.
func (s *SyncService) SyncNow() error {
	s.mu.Lock()
	s.status.Syncing = true
	s.mu.Unlock()
	err := s.engine.FullSync()
	s.mu.Lock()
	s.status.Syncing = false
	s.status.LastError = err
	if err == nil {
		s.status.LastSyncedAt = option.Some(time.Now())
	}
	s.mu.Unlock()
	return err
}

// PushNow runs a push synchronously.
func (s *SyncService) PushNow() error {
	s.mu.Lock()
	s.status.Syncing = true
	s.mu.Unlock()
	err := s.engine.PushChanges()
	s.mu.Lock()
	s.status.Syncing = false
	s.status.LastError = err
	if err == nil {
		s.status.LastSyncedAt = option.Some(time.Now())
	}
	s.mu.Unlock()
	return err
}

// NotifyChange schedules a debounced push; rapid calls collapse into one.
func (s *SyncService) NotifyChange() {
	s.mu.Lock()
	defer s.mu.Unlock()
	if s.timer != nil {
		s.timer.Stop()
	}
	s.timer = time.AfterFunc(s.debounce, func() {
		s.mu.Lock()
		s.status.Syncing = true
		s.mu.Unlock()
		err := s.engine.PushChanges()
		s.mu.Lock()
		s.status.Syncing = false
		s.status.LastError = err
		if err == nil {
			s.status.LastSyncedAt = option.Some(time.Now())
		}
		s.mu.Unlock()
	})
}

func (s *SyncService) Status() SyncStatus {
	s.mu.Lock()
	defer s.mu.Unlock()
	return s.status
}
```

**Grammar (exact):**
1. Trim input. Empty → ErrEmptyTitle.
2. Tokenize by whitespace. Scan left to right:
   - A token starting with `#` starts the list: list name = everything after `#` in that token plus all remaining tokens, joined by spaces, trimmed. Scanning stops.
   - A token starting with `@` starts the date hint: hint tokens = the token with `@` stripped, followed by any consecutive tokens that look like times (`^\d{1,2}:\d{2}$` or `^\d{1,2}\s*(am|pm)$`) or the word `at`. Hint = joined hint tokens; deadline = `dateparser.Parse(hint, now, defaultTime)`.
   - Anything else is a title token.
3. Join title tokens with spaces. Trimmed empty → ErrEmptyTitle.
4. `Deadline` is `None` when no `@` token; `List` is `None` when no `#` token.

// Package quickcapture parses one-line quick-capture strings like
// "buy milk @tomorrow 9:00 #groceries".
package quickcapture

import (
	"errors"
	"regexp"
	"strings"
	"time"

	"github.com/vladgrskkh/psst/internal/dateparser"
	"github.com/vladgrskkh/psst/internal/domain"
)

var ErrEmptyTitle = errors.New("todo title must not be empty")

var (
	reTime    = regexp.MustCompile(`^\d{1,2}:\d{2}$`)
	reTimeAMP = regexp.MustCompile(`^\d{1,2}\s*(am|pm)$`)
)

type Result struct {
	Title    string
	Deadline option.Option[time.Time]
	List     option.Option[string]
}

func isTimeToken(s string) bool {
	return reTime.MatchString(s) || reTimeAMP.MatchString(s)
}

// Parse extracts title, deadline hint and list from a quick-capture string.
func Parse(s string, now time.Time, defaultTime string) (*Result, error) {
	s = strings.TrimSpace(s)
	if s == "" {
		return nil, ErrEmptyTitle
	}
	tokens := strings.Fields(s)
	res := &Result{}
	titleParts := make([]string, 0, len(tokens))

	for i := 0; i < len(tokens); i++ {
		tok := tokens[i]
		switch {
		case strings.HasPrefix(tok, "#"):
			list := strings.TrimSpace(strings.TrimPrefix(tok, "#"))
			if i+1 < len(tokens) {
				list = strings.TrimSpace(list + " " + strings.Join(tokens[i+1:], " "))
			}
			if list != "" {
				res.List = option.Some(list)
			}
			i = len(tokens)
		case strings.HasPrefix(tok, "@"):
			hintParts := []string{strings.TrimPrefix(tok, "@")}
			for i+1 < len(tokens) && (isTimeToken(tokens[i+1]) || tokens[i+1] == "at") {
				i++
				hintParts = append(hintParts, tokens[i])
			}
			hint := strings.Join(hintParts, " ")
			if hint != "" {
				deadline, err := dateparser.Parse(hint, now, defaultTime)
				if err != nil {
					return nil, err
				}
				res.Deadline = option.Some(deadline)
			}
		default:
			titleParts = append(titleParts, tok)
		}
	}

	res.Title = strings.TrimSpace(strings.Join(titleParts, " "))
	if res.Title == "" {
		return nil, ErrEmptyTitle
	}
	return res, nil
}
```

import (

var errUsage = errors.New("usage: psst new \"<text>\" | psst --version")

type systemClock struct{}

func (systemClock) Now() time.Time { return time.Now() }

func run(args []string) (string, error) {
	if len(args) > 1 {
		switch args[1] {
		case "--version":
			return version + "\n", nil
		case "new":
			if len(args) < 3 {
				return "", errUsage
			}
			return runNew(args[2])
		default:
			return "", errUsage
		}
	}
	return "psst: the TUI lands in plan 2 — try `psst new \"buy milk @tomorrow 9:00\"`\n", nil
}

func runNew(text string) (string, error) {
	cfg := loadConfig()

	logger := newLogger(cfg)
	ctx, err := newAppContext(cfg, logger)
	if err != nil {
		return "", err
	}
	defer ctx.close()

	res, err := quickcapture.Parse(text, time.Now(), cfg.General.DefaultTime)
	if err != nil {
		return "", err
	}

	listName := cfg.General.DefaultList
	if res.List.IsPresent() {
		listName = res.List.ValueOr("")
	}

	todo, err := ctx.service.Create(service.CreateTodoParams{
		Title:       res.Title,
		Description: "",
		Deadline:    res.Deadline,
		ListName:    listName,
	})
	if err != nil {
		return "", err
	}

	if err := ctx.sync.SyncNow(); err != nil {
		ctx.logger.Warn("initial sync failed", slog.Any("error", err))
	}
	if err := ctx.sync.PushNow(); err != nil {
		ctx.logger.Warn("push failed", slog.Any("error", err))
	}

	out := fmt.Sprintf("✓ created: %q", todo.Title)
	if todo.Deadline.IsPresent() {
		out += fmt.Sprintf(" — due %s", todo.Deadline.ValueOr(time.Time{}).Format("2006-01-02 15:04"))
	}
	out += fmt.Sprintf(" — list %s", todo.ListName)
	return out + "\n", nil
}

type appContext struct {
	cfg     *config.Config
	logger  *slog.Logger
	db      closer
	service *service.TodoService
	sync    *service.SyncService
}

type closer interface{ Close() error }

func (a *appContext) close() {
	if a.db != nil {
		a.db.Close() //nolint:errcheck
	}
}

func loadConfig() *config.Config {
	path := os.Getenv("PSST_CONFIG")
	if path == "" {
		home, err := os.UserHomeDir()
		if err != nil {
			path = ""
		} else {
			path = filepath.Join(home, ".config", "psst", "config.toml")
		}
	}
	cfg, err := config.Load(path)
	if err != nil {
		fmt.Fprintln(os.Stderr, "psst: config error:", err)
		os.Exit(1)
	}
	return cfg
}

func newLogger(cfg *config.Config) *slog.Logger {
	logPath, err := config.DefaultLogPath()
	if err != nil {
		return slog.New(slog.NewTextHandler(os.Stderr, nil))
	}
	f, err := os.OpenFile(logPath, os.O_WRONLY|os.O_CREATE|os.O_APPEND, 0o644)
	if err != nil {
		return slog.New(slog.NewTextHandler(os.Stderr, nil))
	}
	level := slog.LevelInfo
	if err := level.UnmarshalText([]byte(cfg.General.LogLevel)); err != nil {
		level = slog.LevelInfo
	}
	return slog.New(slog.NewTextHandler(f, &slog.HandlerOptions{Level: level}))
}

func newAppContext(cfg *config.Config, logger *slog.Logger) (*appContext, error) {
	dbPath := cfg.General.DBPath
	if dbPath == "" {
		var err error
		dbPath, err = config.DefaultDBPath()

			return nil, err
		}

		return nil, err

	todoService := service.NewTodoService(todoStore, clock)

		todoStore, todoStore, gateway, clock,

	)
	syncService := service.NewSyncService(engine, time.Duration(cfg.General.PushDebounceMs)*time.Millisecond)
	return &appContext{
		cfg: cfg, logger: logger, db: db, service: todoService, sync: syncService,
	}, nil
